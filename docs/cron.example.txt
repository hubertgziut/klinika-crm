# Przykład: nocny backup Klinika CRM (launchd na macOS)
# Zapisz jako ~/Library/LaunchAgents/pl.klinika.backup.plist i załaduj:
#   launchctl load ~/Library/LaunchAgents/pl.klinika.backup.plist
# Plik .plist:
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE plist PUBLIC "-//Apple//DTD PLIST 1.0//EN" "http://www.apple.com/DTDs/PropertyList-1.0.dtd">
<plist version="1.0">
<dict>
  <key>Label</key><string>pl.klinika.backup</string>
  <key>ProgramArguments</key>
  <array>
    <string>/usr/bin/env</string>
    <string>npm</string>
    <string>run</string>
    <string>backup</string>
  </array>
  <key>WorkingDirectory</key><string>/Users/<UZYTKOWNIK>/Documents/EDMapp/CRM_clinic</string>
  <key>StartCalendarInterval</key>
  <dict>
    <key>Hour</key><integer>2</integer>
    <key>Minute</key><integer>0</integer>
  </dict>
</dict>
</plist>
